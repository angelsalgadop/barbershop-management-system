📍 UBICACIÓN: /root/TU/mobile-app/

✅ COMPILAR APK ONLINE (Sin Android SDK) - 3 PASOS:

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

📝 PASO 1: Crear repositorio en GitHub

   1. Ve a: https://github.com/new
   2. Nombre: barbershop-mobile-app
   3. Puede ser Privado
   4. Crear repositorio

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

🚀 PASO 2: Subir código a GitHub

   cd /root/TU/mobile-app
   
   git config user.email "tu@email.com"
   git config user.name "Tu Nombre"
   
   git remote add origin https://github.com/TU_USUARIO/barbershop-mobile-app.git
   git push -u origin main

   Si pide contraseña, usa Personal Access Token:
   GitHub > Settings > Developer settings > Personal access tokens

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

⚡ PASO 3: Compilar automáticamente

   1. Ve a tu repo en GitHub.com
   2. Clic en "Actions"
   3. Clic en "Build Android APK"
   4. Clic en "Run workflow"
   5. Espera 5-10 minutos ⏰

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

📥 DESCARGAR APK:

   1. Cuando termine (check verde ✓)
   2. Clic en la compilación
   3. Baja a "Artifacts"
   4. Descarga "app-debug"
   5. Extrae el ZIP
   6. ¡Obtendrás app-debug.apk! 🎉

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

📱 INSTALAR:

   1. Copia APK a tu teléfono
   2. Abre el archivo
   3. Habilita "Instalar apps desconocidas"
   4. ¡Instala!
